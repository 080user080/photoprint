# -*- coding: utf-8 -*-

"""
Плагін групових ярликів.
Створює/видаляє папки Друк_<клієнт> з target.lnk та Desktop.ini
на основі вмісту photo_report.txt.
Реагує на зміни звіту, періодичну перевірку mtime та аварійний тригер.
"""

import os
import time
import threading
import traceback
from pathlib import Path
from typing import List, Optional


# ------------------------------------------------------------
# Допоміжні функції для атрибутів
# ------------------------------------------------------------

def _run_attrib(args: List[str], path: str):
    """Запускає attrib.exe з вказаними аргументами для файлу/папки."""
    try:
        import subprocess
        subprocess.run(
            ["attrib"] + args + [path],
            capture_output=True,
            creationflags=0x08000000 if os.name == 'nt' else 0
        )
    except Exception:
        pass


def _set_attrs(path: str, attrs: str):
    """Встановлює атрибути: 'R' (Read-only), 'H' (Hidden), 'S' (System)."""
    args = []
    for ch in attrs.upper():
        if ch == 'R':
            args.append('+R')
        elif ch == 'H':
            args.append('+H')
        elif ch == 'S':
            args.append('+S')
    if args:
        _run_attrib(args, path)


def _remove_attrs(path: str, attrs: str):
    """Знімає атрибути."""
    args = []
    for ch in attrs.upper():
        if ch == 'R':
            args.append('-R')
        elif ch == 'H':
            args.append('-H')
        elif ch == 'S':
            args.append('-S')
    if args:
        _run_attrib(args, path)


# ------------------------------------------------------------
# Робота з файлами
# ------------------------------------------------------------

def create_shortcut(shortcut_dir: str, target_path: str) -> bool:
    """Створює target.lnk у shortcut_dir, що вказує на target_path."""
    try:
        shortcut_file = os.path.join(shortcut_dir, "target.lnk")
        ps_script = f'''
$WScriptShell = New-Object -ComObject WScript.Shell
$Shortcut = $WScriptShell.CreateShortcut('{shortcut_file}')
$Shortcut.TargetPath = '{target_path}'
$Shortcut.Save()
'''
        import subprocess
        subprocess.run(
            ["powershell", "-Command", ps_script],
            capture_output=True,
            creationflags=0x08000000 if os.name == 'nt' else 0
        )
        return os.path.exists(shortcut_file)
    except Exception:
        return False


def create_desktop_ini(folder: str) -> bool:
    """Створює Desktop.ini з необхідним вмістом."""
    try:
        ini_path = os.path.join(folder, "Desktop.ini")
        content = (
            "[.ShellClassInfo]\n"
            "CLSID2={0AFACED1-E828-11D1-9187-B532F1E9575D}\n"
            "Flags=2\n"
            "ConfirmFileOp=0\n"
        )
        # Видаляємо старий, якщо є
        if os.path.exists(ini_path):
            _remove_attrs(ini_path, "RHS")
            os.remove(ini_path)
        with open(ini_path, 'w', encoding='utf-8') as f:
            f.write(content)
        # Атрибути: H + S
        _set_attrs(ini_path, "HS")
        return True
    except Exception:
        return False


def create_client_folder(client: str, output_base: str, folder_prefix: str,
                         session_target: str) -> Optional[str]:
    """
    Створює (оновлює) папку Друк_<client>, ярлик target.lnk та Desktop.ini.
    Повертає шлях до папки, якщо успіх, інакше None.
    """
    folder_path = os.path.join(output_base, f"{folder_prefix}{client}")
    try:
        os.makedirs(folder_path, exist_ok=True)
        # Знімаємо атрибути на час оновлення
        _remove_attrs(folder_path, "RS")
        # Ярлик
        create_shortcut(folder_path, session_target)
        # Desktop.ini
        create_desktop_ini(folder_path)
        # Встановлюємо R+S на папку
        _set_attrs(folder_path, "RS")
        return folder_path
    except Exception:
        return None


def remove_client_folder(client: str, output_base: str, folder_prefix: str):
    """Видаляє папку Друк_<client> разом із вмістом."""
    folder_path = os.path.join(output_base, f"{folder_prefix}{client}")
    if not os.path.isdir(folder_path):
        return
    try:
        # Знімаємо атрибути з папки та її вмісту
        _remove_attrs(folder_path, "RS")
        for name in ("target.lnk", "Desktop.ini"):
            fp = os.path.join(folder_path, name)
            if os.path.exists(fp):
                _remove_attrs(fp, "RHS")
        import shutil
        shutil.rmtree(folder_path)
    except Exception:
        pass

# Синхронізація за вмістом звіту
# ------------------------------------------------------------

def parse_report_lines(lines: List[str]):
    """
    Зі списку рядків звіту повертає множину клієнтів та
    словник: client -> session_target_path (найглибша сесія).
    Використовує перші 4 компоненти відносного шляху як сесійну папку.
    """
    clients = set()
    client_session = {}  # client -> session_path (абсолютний або відносний)

    # Тут ми не знаємо watch_path, тому зберігаємо лише ім'я клієнта та сесійний відносний шлях.
    # При створенні ярлика потрібно буде з'єднати з watch_path. Але плагін не має прямого доступу
    # до watch_path, тому зробимо інакше: збережемо список (client, relative_session_path).
    # Але для уніфікації краще повертати client -> relative_session_path.
    # Однак для ярлика потрібен абсолютний шлях. Плагін отримає config.watch_paths.
    # Тому в sync_shortcuts будемо передавати config.
    # У цій функції просто повернемо клієнтів та session_relative.
    # Для спрощення викликатимемо цю логіку в sync_shortcuts безпосередньо.
    # Але для повторного використання напишемо окремо.
    for line in lines:
        line = line.strip()
        if not line or ' -- ' not in line:
            continue
        parts = line.split(' -- ')
        if len(parts) < 2:
            continue
        relative_path = parts[1]
        components = relative_path.split(os.sep)
        if not components:
            continue
        client = components[0]
        clients.add(client)
        # Беремо перші 4 компоненти (клієнт + рік + місяць + сесія)
        session_components = components[:min(4, len(components))]
        session_rel = os.sep.join(session_components)
        # Для кожного клієнта зберігаємо останню (за порядком у звіті) сесію
        client_session[client] = session_rel
    return clients, client_session


def sync_shortcuts(config, lines: List[str]):
    """
    Синхронізує групові ярлики з поточним вмістом звіту.
    Створює/оновлює папки для присутніх клієнтів,
    видаляє для відсутніх.
    """
    if not config.group_shortcuts_enabled:
        return

    clients_in_report, client_session_rel = parse_report_lines(lines)

    # Визначаємо watch_path (беремо перший існуючий, або як раніше)
    watch_base = None
    for wp in config.watch_paths:
        if os.path.isdir(wp):
            watch_base = os.path.normpath(wp)
            break
    if not watch_base:
        return

    # Обробляємо клієнтів зі звіту
    for client, session_rel in client_session_rel.items():
        target_session = os.path.join(watch_base, session_rel)
        create_client_folder(client, config.group_shortcuts_output_base,
                             config.group_shortcuts_folder_prefix,
                             target_session)

    # Видаляємо папки клієнтів, яких немає у звіті
    output_base = config.group_shortcuts_output_base
    if not os.path.isdir(output_base):
        return
    prefix = config.group_shortcuts_folder_prefix
    for entry in os.listdir(output_base):
        if not entry.startswith(prefix):
            continue
        client_name = entry[len(prefix):]
        if client_name not in clients_in_report:
            remove_client_folder(client_name, output_base, prefix)


# ------------------------------------------------------------
# Фонова перевірка mtime та аварійного тригера
# ------------------------------------------------------------

class SyncWatcher(threading.Thread):
    """Періодично перевіряє зміну звіту за mtime та наявність аварійного файлу."""

    def __init__(self, config, sync_func, log_func=None, error_log_func=None):
        super().__init__(daemon=True)
        self.config = config
        self.sync_func = sync_func          # callable(config, lines)
        self.log = log_func
        self.err = error_log_func
        self._stop_event = threading.Event()
        self._last_mtime = None
        self._report_path = config.output_txt

    def run(self):
        # Початкова синхронізація (на випадок ручних змін до старту)
        self._check_and_sync()
        # Отримуємо початковий mtime
        self._update_mtime()

        while not self._stop_event.wait(timeout=self.config.report_check_interval):
            try:
                # Перевірка аварійного тригера (частіше)
                trigger_file = self.config.emergency_trigger_file
                if trigger_file and os.path.exists(trigger_file):
                    if self.log:
                        self.log(f"group_shortcuts: знайдено аварійний файл {trigger_file}")
                    self._check_and_sync()
                    # Видаляємо тригер
                    try:
                        os.remove(trigger_file)
                    except:
                        pass
                # Перевірка зміни звіту за mtime
                current_mtime = self._get_mtime()
                if current_mtime is not None and current_mtime != self._last_mtime:
                    if self.log:
                        self.log("group_shortcuts: виявлено зміну звіту за mtime")
                    self._check_and_sync()
                    self._last_mtime = current_mtime
            except Exception as e:
                if self.err:
                    self.err(f"group_shortcuts: помилка у вотчері: {e}")

    def _check_and_sync(self):
        try:
            lines = self._read_lines()
            self.sync_func(self.config, lines)
        except Exception as e:
            if self.err:
                self.err(f"group_shortcuts: помилка синхронізації: {e}")

    def _read_lines(self):
        if not os.path.exists(self._report_path):
            return []
        for enc in [self.config.output_encoding, 'utf-8', 'cp1251']:
            try:
                with open(self._report_path, 'r', encoding=enc) as f:
                    return f.readlines()
            except:
                continue
        return []

    def _get_mtime(self):
        if os.path.exists(self._report_path):
            return os.path.getmtime(self._report_path)
        return None

    def _update_mtime(self):
        self._last_mtime = self._get_mtime()

    def stop(self):
        self._stop_event.set()


# ------------------------------------------------------------
# Реєстрація плагіна
# ------------------------------------------------------------

def register(cb, config):
    """Підписуємось на події callback_manager."""
    # Створюємо вотчер
    watcher = SyncWatcher(
        config,
        sync_shortcuts,
        log_func=_plugin_log,
        error_log_func=_plugin_err
    )

    # Підписка на оновлення звіту (реактивна)
    def on_report_update(lines):
        sync_shortcuts(config, lines)

    cb.on_report_updated.append(on_report_update)

    # Старт вотчера при запуску
    def on_start():
        watcher.start()

    cb.on_startup.append(on_start)

    # Зупинка при завершенні
    def on_stop():
        watcher.stop()

    cb.on_shutdown.append(on_stop)


# Приватні логери плагіна (можна ігнорувати, помилки пишуться в основний лог)
def _plugin_log(msg):
    try:
        with open("monitor_events.log", 'a', encoding='utf-8') as f:
            from datetime import datetime
            f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {msg}\n")
    except:
        pass


def _plugin_err(msg):
    try:
        with open("monitor_errors.log", 'a', encoding='utf-8') as f:
            from datetime import datetime
            f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {msg}\n")
    except:
        pass

# -*- coding: utf-8 -*-

"""
Менеджер зворотних викликів (хуків) для плагінів.
Надає події: on_startup, on_shutdown, on_report_updated.
"""

from typing import List, Callable


class CallbackManager:
    """Збирає обробники подій та викликає їх у порядку реєстрації."""

    def __init__(self):
        self.on_startup: List[Callable[[], None]] = []
        self.on_shutdown: List[Callable[[], None]] = []
        self.on_report_updated: List[Callable[[List[str]], None]] = []

    def startup(self):
        for handler in self.on_startup:
            try:
                handler()
            except Exception:
                pass  # логування за потреби

    def shutdown(self):
        for handler in self.on_shutdown:
            try:
                handler()
            except Exception:
                pass

    def report_updated(self, lines: List[str]):
        for handler in self.on_report_updated:
            try:
                handler(lines)
            except Exception:
                pass

# -*- coding: utf-8 -*-

"""
Менеджер очищення: порядкове видалення старих записів звіту, логів, файлів-прапорців.
Більше не керує груповими ярликами – це робить окремий плагін.
"""

import os
import time
import threading
from datetime import datetime, timedelta
from pathlib import Path
from typing import List, Optional


class CleanupManager:
    """Керує очищенням старих звітів та файлів-прапорців."""

    def __init__(self, config):
        self.config = config
        self.running = False
        self.thread: Optional[threading.Thread] = None

    def start(self):
        if self.running:
            return
        self.running = True
        self.thread = threading.Thread(target=self._cleanup_loop, daemon=True)
        self.thread.start()
        self._log_event("Менеджер очищення запущено")

    def stop(self):
        self.running = False
        if self.thread:
            self.thread.join(timeout=5)
        self._log_event("Менеджер очищення зупинено")

    def _cleanup_loop(self):
        last_cleanup = datetime.now()
        while self.running:
            try:
                now = datetime.now()
                hours_passed = (now - last_cleanup).total_seconds() / 3600
                if hours_passed >= self.config.cleanup_period_hours:
                    self._perform_cleanup()
                    last_cleanup = now
                time.sleep(self.config.cleanup_check_interval)
            except Exception as e:
                self._log_event(f"Помилка в циклі очищення: {e}")
                time.sleep(300)

    def _perform_cleanup(self):
        try:
            self._log_event("Початок очищення")
            self._cleanup_report_by_lines()
            if self.config.create_flag_file:
                self._cleanup_flag_files()
            self._cleanup_old_logs()
            self._log_event("Очищення завершено")
        except Exception as e:
            self._log_event(f"Помилка під час очищення: {e}")

    def _cleanup_report_by_lines(self):
        """Видаляє рядки звіту, старші за log_retention_days, без зміни формату дати."""
        report_path = Path(self.config.output_txt)
        if not report_path.exists():
            return

        try:
            encodings = [self.config.output_encoding, 'utf-8', 'cp1251']
            lines = []
            used_enc = None
            for enc in encodings:
                try:
                    with open(report_path, 'r', encoding=enc) as f:
                        lines = f.readlines()
                    used_enc = enc
                    break
                except (UnicodeDecodeError, LookupError):
                    continue
            if used_enc is None:
                self._log_event("Не вдалося прочитати звіт у жодному з кодувань, очищення пропущено")
                return

            today = datetime.now().date()
            retention_days = self.config.log_retention_days
            new_lines = []
            removed = 0

            for line in lines:
                stripped = line.rstrip('\n\r')
                if not stripped:
                    new_lines.append(line)
                    continue

                if ' -- ' not in stripped:
                    new_lines.append(line)
                    continue

                date_time_part = stripped.split(' -- ')[0]
                if '=' not in date_time_part:
                    new_lines.append(line)
                    continue

                date_str = date_time_part.split('=')[0].strip()
                try:
                    day, month = map(int, date_str.split('-'))
                except (ValueError, AttributeError):
                    new_lines.append(line)
                    continue

                parsed_date = self._resolve_date(day, month, today)
                if parsed_date is None:
                    removed += 1
                    continue

                age = (today - parsed_date).days
                if age > retention_days:
                    removed += 1
                else:
                    new_lines.append(line)

            if removed > 0:
                with open(report_path, 'w', encoding=used_enc) as f:
                    f.writelines(new_lines)
                self._log_event(f"Зі звіту видалено {removed} старих рядків (залишено {len(new_lines)}).")
            else:
                self._log_event("Немає рядків для видалення.")

        except Exception as e:
            self._log_event(f"Помилка при порядковому очищенні звіту: {e}")

    def _resolve_date(self, day: int, month: int, today):
        current_year = today.year
        try:
            d = datetime(current_year, month, day).date()
        except ValueError:
            for y in range(current_year - 1, current_year - 5, -1):
                try:
                    d = datetime(y, month, day).date()
                    break
                except ValueError:
                    continue
            else:
                return None

        if d > today:
            try:
                d = datetime(current_year - 1, month, day).date()
            except ValueError:
                for y in range(current_year - 2, current_year - 6, -1):
                    try:
                        d = datetime(y, month, day).date()
                        break
                    except ValueError:
                        continue
                else:
                    return None
        return d

    def _cleanup_flag_files(self):
        """Видаляє старі .new файли."""
        try:
            cutoff = datetime.now() - timedelta(days=self.config.flag_retention_days)
            deleted = 0
            for f in Path('.').glob('*.new'):
                if f.is_file():
                    ctime = datetime.fromtimestamp(f.stat().st_ctime)
                    if ctime < cutoff:
                        f.unlink()
                        deleted += 1
            if deleted:
                self._log_event(f"Видалено {deleted} старих файлів-прапорців")
        except Exception as e:
            self._log_event(f"Помилка очищення прапорців: {e}")

    def _cleanup_old_logs(self):
        """Видаляє дуже старі лог-файли."""
        try:
            log_files = ["monitor_errors.log", "monitor_events.log", "cleanup_events.log"]
            cutoff = datetime.now() - timedelta(days=self.config.log_retention_days * 2)
            for name in log_files:
                p = Path(name)
                if p.exists():
                    mtime = datetime.fromtimestamp(p.stat().st_mtime)
                    if mtime < cutoff:
                        p.unlink()
                        self._log_event(f"Видалено старий лог: {name}")
        except Exception as e:
            self._log_event(f"Помилка очищення логів: {e}")

    def _log_event(self, message: str):
        try:
            with open("cleanup_events.log", 'a', encoding='utf-8') as f:
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {message}\n")
        except:
            pass

# -*- coding: utf-8 -*-

"""
Конфігурація моніторингу. Завантаження з .ini, створення стандартного файлу.
"""

import os
import configparser
from dataclasses import dataclass, field
from typing import List


@dataclass
class MonitorConfig:
    """Зберігає конфігурацію моніторингу."""
    watch_paths: List[str]
    output_txt: str
    extensions_photo: List[str]
    extensions_extra: List[str]
    recursive: bool
    scan_delay_seconds: int
    count_subfolders: bool
    cleanup_period_hours: int
    cleanup_check_interval: int
    log_retention_days: int
    create_flag_file: bool
    flag_retention_days: int
    output_encoding: str
    enable_logging: bool

    # Налаштування ярликів (копіювання в нові папки)

"""
Обробник подій файлової системи: виявлення нових папок, аналіз вмісту, запис у звіт.
Після кожної зміни звіту сповіщає CallbackManager.
"""

import os
import threading
import traceback
from datetime import datetime
from pathlib import Path
from typing import Dict, Set, Optional, List

from watchdog.events import FileSystemEventHandler

from .shortcuts import copy_shortcuts
# Видалено прямий імпорт group_shortcuts – тепер через плагін


class NewFolderHandler(FileSystemEventHandler):
    """Обробляє події створення папок та файлів."""

    def __init__(self, config, processed_lock: threading.Lock, callback_manager=None):
        super().__init__()
        self.config = config
        self.processed_folders: Set[str] = set()
        self.processed_lock = processed_lock
        self.pending_folders: Dict[str, threading.Timer] = {}
        self.report_lock = threading.Lock()
        self.callback_manager = callback_manager  # нове

    def on_created(self, event):
        """Викликається при створенні файлу або папки."""
        try:
            if self.config.enable_logging:
                self._log_event(f"Подія створення: {event.src_path} (каталог: {event.is_directory})")

            if not event.is_directory:
                parent_folder = os.path.dirname(event.src_path)
                if self.config.enable_logging:
                    self._log_event(f"Файл створено: {event.src_path}, перевіряємо папку: {parent_folder}")
                if parent_folder not in self.pending_folders:
                    self._schedule_folder_check(parent_folder)
            else:
                if self.config.enable_logging:
                    self._log_event(f"Папка створена: {event.src_path}")
                self._handle_new_folder(event.src_path)
        except Exception as e:
            self._log_error(f"Помилка в on_created: {e}\n{traceback.format_exc()}")

    def _handle_new_folder(self, folder_path: str):
        """Обробляє нову папку."""
        normalized_path = os.path.normpath(folder_path)

        with self.processed_lock:
            if normalized_path in self.processed_folders:
                if self.config.enable_logging:
                    self._log_event(f"Папка вже оброблялась: {normalized_path}")
                return
            self.processed_folders.add(normalized_path)

        if self.config.enable_logging:
            self._log_event(f"Нова папка для обробки: {normalized_path}")

        self._schedule_folder_check(normalized_path)

    def _schedule_folder_check(self, folder_path: str):
        """Планує аналіз папки через заданий час."""
        try:
            if folder_path in self.pending_folders:
                self.pending_folders[folder_path].cancel()

            timer = threading.Timer(
                self.config.scan_delay_seconds,
                self._analyze_folder,
                args=(folder_path,)
            )
            timer.daemon = True
            self.pending_folders[folder_path] = timer
            timer.start()

            if self.config.enable_logging:
                self._log_event(f"Заплановано перевірку папки через {self.config.scan_delay_seconds} сек: {folder_path}")
        except Exception as e:
            self._log_error(f"Помилка планування перевірки папки {folder_path}: {e}")

    def _analyze_folder(self, folder_path: str):
        """Аналізує вміст папки та формує запис для звіту."""
        try:
            if folder_path in self.pending_folders:
                del self.pending_folders[folder_path]

            if not os.path.exists(folder_path) or not os.path.isdir(folder_path):
                if self.config.enable_logging:
                    self._log_event(f"Папка не існує або не є каталогом: {folder_path}")
                return

            if self.config.enable_logging:
                self._log_event(f"Початок аналізу папки: {folder_path}")

            counts = self._count_files_in_folder(folder_path)
            if self.config.enable_logging:
                self._log_event(f"Результат підрахунку для {folder_path}: {counts}")

            if any(counts.values()):
                self._write_to_report(folder_path, counts)
                # Створюємо файл-прапорець після успішного запису
                if self.config.create_flag_file:
                    self._create_flag_file()

                # Запускаємо відкладене копіювання ярликів (звичайних, не групових)
                if self.config.shortcut_source_paths:
                    self._schedule_shortcut_copy(folder_path)
            else:
                if self.config.enable_logging:
                    self._log_event(f"У папці {folder_path} немає відповідних файлів")
        except Exception as e:
            self._log_error(f"Помилка при аналізі папки {folder_path}: {e}\n{traceback.format_exc()}")

    def _count_files_in_folder(self, folder_path: str) -> Dict[str, int]:
        """Рахує кількість файлів за типами у папці."""
        counts = {}
        all_extensions = self.config.extensions_photo + self.config.extensions_extra
        for ext in all_extensions:
            counts[ext] = 0

        try:
            if self.config.count_subfolders:
                for root, dirs, files in os.walk(folder_path):
                    for file in files:
                        ext = os.path.splitext(file)[1].lower().lstrip('.')
                        if ext in counts:
                            counts[ext] += 1
            else:
                for item in os.listdir(folder_path):
                    item_path = os.path.join(folder_path, item)
                    if os.path.isfile(item_path):
                        ext = os.path.splitext(item)[1].lower().lstrip('.')
                        if ext in counts:
                            counts[ext] += 1
        except (PermissionError, OSError) as e:
            self._log_error(f"Помилка доступу до {folder_path}: {e}")
        except Exception as e:
            self._log_error(f"Несподівана помилка при підрахунку файлів у {folder_path}: {e}")

        return counts

    def _get_display_path(self, folder_path: str) -> str:

        folder_path = os.path.normpath(folder_path)
        for base_path in self.config.watch_paths:
            normalized_base = os.path.normpath(base_path)
            if folder_path.lower().startswith(normalized_base.lower()):
                relative = folder_path[len(normalized_base):]
                return relative.strip(os.sep)
        return folder_path

    def _get_watch_path_for_folder(self, folder_path: str) -> Optional[str]:
        """Повертає базовий шлях з watch_paths, якому належить folder_path."""
        folder_path = os.path.normpath(folder_path)
        for base in self.config.watch_paths:
            norm_base = os.path.normpath(base)
            if folder_path.lower().startswith(norm_base.lower()):
                return norm_base
        return None

    def _format_report_line(self, folder_path: str, counts: Dict[str, int]) -> Optional[str]:
        """Формує рядок звіту без дати/часу."""
        display_path = self._get_display_path(folder_path)
        photo_count = sum(counts[ext] for ext in self.config.extensions_photo)

        parts = []
        if photo_count == 1:
            parts.append("1 ФОТО")
        elif photo_count > 1:
            parts.append(f"{photo_count} ФОТО")

        for ext in self.config.extensions_extra:
            cnt = counts.get(ext, 0)
            if cnt > 0:
                if cnt == 1:
                    parts.append(f"1 файл .{ext}")
                else:
                    parts.append(f"{cnt} файли .{ext}")

        if not parts:
            return None
        return f"{display_path} -- {', '.join(parts)}"

    def _update_or_append_record(self, date_str: str, time_str: str, line_content: str):
        """Оновлює існуючий запис для цієї папки за поточну дату або додає новий."""
        with self.report_lock:
            report_path = self.config.output_txt
            full_line = f"{date_str}={time_str} -- {line_content}\n"

            try:
                if not os.path.exists(report_path):
                    with open(report_path, 'w', encoding=self.config.output_encoding) as f:
                        f.write(full_line)
                    if self.config.enable_logging:
                        self._log_event(f"Створено новий звіт із записом: {full_line.strip()}")
                    return

                try:
                    with open(report_path, 'r', encoding=self.config.output_encoding) as f:
                        lines = f.readlines()
                except UnicodeDecodeError:
                    with open(report_path, 'r', encoding='utf-8') as f:
                        lines = f.readlines()

                new_lines = []
                found = False
                display_path = line_content.split(" -- ")[0]

                for line in lines:
                    stripped = line.strip()
                    parts = stripped.split(" -- ", 2)
                    if len(parts) >= 2:
                        date_time_part = parts[0]
                        path_part = parts[1]
                        if date_time_part.startswith(f"{date_str}=") and path_part == display_path:
                            new_lines.append(full_line)
                            found = True
                            if self.config.enable_logging:
                                self._log_event(f"Оновлено запис для папки: {display_path}")
                            continue
                    new_lines.append(line)

                if not found:
                    new_lines.append(full_line)
                    if self.config.enable_logging:
                        self._log_event(f"Додано новий запис для папки: {display_path}")

                with open(report_path, 'w', encoding=self.config.output_encoding) as f:
                    f.writelines(new_lines)

            except Exception as e:
                self._log_error(f"Помилка при оновленні запису у звіті: {e}\n{traceback.format_exc()}")
                try:
                    with open(report_path, 'a', encoding='utf-8') as f:
                        f.write(full_line)
                except:
                    pass

            if not os.path.exists(self.config_path):
                MonitorConfig.create_default_config(self.config_path)
                self._log_fatal_error(f"Створено стандартний конфіг: {self.config_path}. Налаштуйте його.")
                return False

            self.config = MonitorConfig.from_ini(self.config_path)

            if not self.config.watch_paths:
                self._log_fatal_error("Не вказано жодного шляху для моніторингу (watch_paths)")
                return False

            valid_paths = []
            for path in self.config.watch_paths:
                if os.path.exists(path):
                    valid_paths.append(path)
                else:
                    self._log_fatal_error(f"Шлях не існує: {path}")

            if not valid_paths:
                self._log_fatal_error("Жоден з вказаних шляхів не існує")
                return False

            self.config.watch_paths = valid_paths
            return True
        except Exception as e:
            self._log_fatal_error(f"Помилка завантаження конфігурації: {e}\n{traceback.format_exc()}")
            return False

    def load_plugins(self):
        """Завантажує плагіни з plugins_dir (відносно робочої директорії або _internal)."""
        if not self.config:
            return
        plugins_dir = self.config.plugins_dir
        if not os.path.isabs(plugins_dir):
            base_dir = os.getcwd()
            candidate = os.path.abspath(os.path.join(base_dir, plugins_dir))
            if not os.path.isdir(candidate):
                # Спробувати всередині _internal/ (куди PyInstaller пакує .py)
                candidate = os.path.abspath(os.path.join(base_dir, "_internal", plugins_dir))
            plugins_dir = candidate

        if not os.path.isdir(plugins_dir):
            if self.config.enable_logging:
                self._log_event(f"Директорія плагінів не знайдена: {plugins_dir}")
            return

        for fn in sorted(os.listdir(plugins_dir)):
            if fn.endswith('.py') and not fn.startswith('_'):
                mod_name = fn[:-3]
                try:
                    mod = importlib.import_module(f"monitor.plugins.{mod_name}")
                    if hasattr(mod, 'register'):
                        mod.register(self.callback_manager, self.config)
                        if self.config.enable_logging:
                            self._log_event(f"Плагін завантажено: {mod_name}")
                except Exception as e:
                    if self.config.enable_logging:
                        self._log_error(f"Помилка завантаження плагіна {mod_name}: {e}")

    def start_monitoring(self):
        if not self.load_configuration():
            time.sleep(3)
            return

        # Завантажуємо плагіни
        self.load_plugins()
        # Викликаємо on_startup для всіх плагінів
        self.callback_manager.startup()

        try:
            self.event_handler = NewFolderHandler(
                self.config,
                self.processed_lock,
                callback_manager=self.callback_manager
            )
            self.observer = Observer()

            for path in self.config.watch_paths:
                try:
                    self.observer.schedule(
                        self.event_handler,
                        path,
                        recursive=self.config.recursive
                    )
                    if self.config.enable_logging:
                        self._log_event(f"Додано шлях для моніторингу: {path}")
                except Exception as e:
                    self._log_error(f"Помилка додавання шляху {path}: {e}")

            self.cleanup_manager = CleanupManager(self.config)
            self.cleanup_manager.start()

            self.observer.start()
            self.running = True

            with open('monitor_running.flag', 'w') as f:
                f.write(f"Моніторинг запущено: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")

            print("Моніторинг запущено. Робота у фоновому режимі.")

            while self.running and self.observer.is_alive():
                time.sleep(1)

        except KeyboardInterrupt:
            print("\nОтримано сигнал зупинки...")
        except Exception as e:
            self._log_fatal_error(f"Критична помилка при запуску моніторингу: {e}\n{traceback.format_exc()}")
        finally:
            self.stop_monitoring()

    def stop_monitoring(self):
        self.running = False
        self.callback_manager.shutdown()
        if os.path.exists('monitor_running.flag'):
            try:
                os.remove('monitor_running.flag')
            except:
                pass

        if self.observer:
            try:
                self.observer.stop()
                self.observer.join(timeout=10)
            except:
                pass

        if self.cleanup_manager:
            try:
                self.cleanup_manager.stop()
            except:
                pass

        print("Моніторинг зупинено.")

    def _log_fatal_error(self, message: str):
        try:
            with open("fatal_error.log", 'a', encoding='utf-8') as f:
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {message}\n")
        except:
            pass
        print(message)

    def _log_event(self, message: str):
        try:
            with open("monitor_events.log", 'a', encoding='utf-8') as f:
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {message}\n")
        except:
            pass

    def _log_error(self, message: str):
        try:
            with open("monitor_errors.log", 'a', encoding='utf-8') as f:
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {message}\n")

            pass

# -*- coding: utf-8 -*-

"""
Копіювання структур ярликів у цільову папку (shutil.copytree).
"""

import os
import shutil
from typing import List, Callable, Optional


def copy_shortcuts(
    source_paths: List[str],
    target_folder: str,
    overwrite: bool = True,
    on_folder_created: Optional[Callable[[str], None]] = None,
    log_func: Optional[Callable[[str], None]] = None,
    error_log_func: Optional[Callable[[str], None]] = None,
):
    """
    Копіює вміст кожної вихідної папки з source_paths у target_folder.
    Якщо target_folder не існує, створює її та викликає on_folder_created.
    """
    if not source_paths:
        return

    # Перевіряємо/створюємо цільову папку
    if not os.path.exists(target_folder):
        try:
            os.makedirs(target_folder, exist_ok=True)
            if log_func:
                log_func(f"Створено відсутню цільову папку: {target_folder}")
            if on_folder_created:
                on_folder_created(target_folder)
        except Exception as e:
            if error_log_func:
                error_log_func(f"Не вдалося створити папку {target_folder}: {e}")
            return

    for src in source_paths:
        if not os.path.isdir(src):
            if log_func:
                log_func(f"Джерело ярликів не існує або не є папкою: {src}")
            continue

        try:
            # copytree з dirs_exist_ok=True копіює вміст src всередину target_folder
            shutil.copytree(src, target_folder, dirs_exist_ok=True)
            if log_func:
                log_func(f"Ярлики з {src} успішно скопійовано в {target_folder}")
        except Exception as e:
            if error_log_func:
                error_log_func(f"Помилка копіювання ярликів з {src} у {target_folder}: {e}")
